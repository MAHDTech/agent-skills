```
import xai_sdk

client = xai_sdk.Client()

response = client.video.generate(
    prompt="A glowing crystal-powered rocket launching from Mars",
    model="grok-imagine-video",
    duration=10,
    aspect_ratio="16:9",
    resolution="720p",
)

print(response.video.url)
```

```
# 1. Start generation
curl -X POST https://api.x.ai/v1/videos/generations \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $XAI_API_KEY" \
  -d '{
    "model": "grok-imagine-video",
    "prompt": "A glowing crystal-powered rocket launching from Mars"
  }'

# 2. Poll with the returned request_id
curl -X GET "https://api.x.ai/v1/videos/REQUEST_ID" \
  -H "Authorization: Bearer $XAI_API_KEY"
```

```
import OpenAI from "openai";

const client = new OpenAI({
    apiKey: process.env.XAI_API_KEY,
    baseURL: "https://api.x.ai/v1",
});

// 1. Start generation
const { request_id } = await client.post("/videos/generations", {
    body: {
        model: "grok-imagine-video",
        prompt: "A glowing crystal-powered rocket launching from Mars",
    },
});

// 2. Poll until done
let result;
do {
    await new Promise((r) => setTimeout(r, 5000));
    result = await client.get(`/videos/${request_id}`);
} while (result.status !== "done");

console.log(result.video.url);
```

```
import time
from openai import OpenAI

client = OpenAI(
    base_url="https://api.x.ai/v1",
    api_key="YOUR_API_KEY",
)

# 1. Start generation
response = client.post(
    "/videos/generations",
    body={
        "model": "grok-imagine-video",
        "prompt": "A glowing crystal-powered rocket launching from Mars",
    },
)
request_id = response["request_id"]

# 2. Poll until done
while True:
    result = client.get(f"/videos/{request_id}")
    if result["status"] == "done":
        print(result["video"]["url"])
        break
    time.sleep(5)
```
